<body>
    <div class="container">
        <h1> Be + Prepositions Game</h1>
        <div class="score">Score: <span id="score">0</span> / <span id="total">0</span></div>
        <div id="game">
            <div class="question" id="question"></div>
            <div class="options" id="options"></div>
        </div>
        <div id="finalResult" style="display:none;"></div>
    </div>

    <script>
        const questions = [
            { sentence: "She is good ___ mathematics", correct: "at", options: ["at", "in", "on", "of"] },
            { sentence: "He is fond ___ classical music", correct: "of", options: ["of", "about", "with", "for"] },
            { sentence: "They are interested ___ learning English", correct: "in", options: ["in", "on", "at", "of"] },
            { sentence: "I am tired ___ doing homework", correct: "of", options: ["of", "from", "with", "about"] },
            { sentence: "She is afraid ___ spiders", correct: "of", options: ["of", "from", "with", "about"] },
            { sentence: "He is keen ___ football", correct: "on", options: ["on", "in", "at", "of"] },
            { sentence: "We are proud ___ our son", correct: "of", options: ["of", "for", "about", "with"] },
            { sentence: "She is married ___ a doctor", correct: "to", options: ["to", "with", "for", "of"] },
            { sentence: "He is responsible ___ the project", correct: "for", options: ["for", "of", "about", "with"] },
            { sentence: "I am bored ___ this movie", correct: "with", options: ["with", "of", "from", "about"] },
            { sentence: "She is famous ___ her paintings", correct: "for", options: ["for", "of", "about", "with"] },
            { sentence: "He is addicted ___ video games", correct: "to", options: ["to", "with", "for", "of"] },
            { sentence: "They are pleased ___ the results", correct: "with", options: ["with", "of", "about", "for"] },
            { sentence: "She is worried ___ the exam", correct: "about", options: ["about", "of", "for", "with"] },
            { sentence: "He is bad ___ remembering names", correct: "at", options: ["at", "in", "on", "of"] }
        ];

        let currentQuestion = 0;
        let score = 0;
        let shuffledQuestions = [...questions].sort(() => Math.random() - 0.5);

        function showQuestion() {
            if (currentQuestion >= shuffledQuestions.length) {
                showFinalResult();
                return;
            }

            const q = shuffledQuestions[currentQuestion];
            document.getElementById('question').textContent = q.sentence;
            document.getElementById('score').textContent = score;
            document.getElementById('total').textContent = shuffledQuestions.length;

            const optionsDiv = document.getElementById('options');
            optionsDiv.innerHTML = '';

            const shuffledOptions = [...q.options].sort(() => Math.random() - 0.5);
            
            shuffledOptions.forEach(option => {
                const btn = document.createElement('button');
                btn.textContent = option;
                btn.onclick = () => checkAnswer(option, q.correct, btn);
                optionsDiv.appendChild(btn);
            });
        }

        function checkAnswer(selected, correct, btn) {
            const buttons = document.querySelectorAll('button');
            buttons.forEach(b => b.disabled = true);

            if (selected === correct) {
                btn.classList.add('correct');
                score++;
            } else {
                btn.classList.add('wrong');
                buttons.forEach(b => {
                    if (b.textContent === correct) b.classList.add('correct');
                });
            }

            setTimeout(() => {
                currentQuestion++;
                showQuestion();
            }, 1500);
        }

        function showFinalResult() {
            document.getElementById('game').style.display = 'none';
            const resultDiv = document.getElementById('finalResult');
            resultDiv.style.display = 'block';
            
            let message = '';
            const percentage = (score / shuffledQuestions.length) * 100;
            
            if (percentage === 100) message = '🏆 Excellent! Perfect score!';
            else if (percentage >= 80) message = '🌟 Great job!';
            else if (percentage >= 60) message = '👍 Good effort!';
            else message = '📖 Keep practicing!';

            resultDiv.innerHTML = `
                <div class="result">
                    <h2>Game Over!</h2>
                    <p>Your score: ${score} out of ${shuffledQuestions.length}</p>
                    <p>${message}</p>
                    <button onclick="location.reload()">Play Again</button>
                </div>
            `;
        }

        showQuestion();
    </script>
