<script>

  const vocabulary = [

    { en: "ambitious", ru: "амбициозный" },

    { en: "generous", ru: "щедрый" },

    { en: "stubborn", ru: "упрямый" },

    { en: "patient", ru: "терпеливый" },

    { en: "selfish", ru: "эгоистичный" },

    { en: "reliable", ru: "надёжный" },

    { en: "sensitive", ru: "чувствительный" },

    { en: "modest", ru: "скромный" },

    { en: "talkative", ru: "разговорчивый" },

    { en: "easy-going", ru: "покладистый" },

    { en: "strict", ru: "строгий" },

    { en: "sincere", ru: "искренний" },

    { en: "competitive", ru: "соревновательный" },

    { en: "polite", ru: "вежливый" },

    { en: "arrogant", ru: "высокомерный" },

    { en: "anxious", ru: "тревожный" },

    { en: "broad-minded", ru: "с широким кругозором" },

    { en: "reserved", ru: "сдержанный" }

  ];


  const TOTAL_QUESTIONS = 10;

  let questions = [];

  let currentIndex = 0;

  let score = 0;

  let answered = false;


  function shuffle(arr) {

    const a = [...arr];

    for (let i = a.length - 1; i > 0; i--) {

      const j = Math.floor(Math.random() * (i + 1));

      [a[i], a[j]] = [a[j], a[i]];

    }

    return a;

  }


  function generateQuestions() {

    const shuffled = shuffle(vocabulary).slice(0, TOTAL_QUESTIONS);

    return shuffled.map(item => {

      // 2 дистрактора из других слов

      const distractors = shuffle(vocabulary.filter(v => v.en !== item.en))

        .slice(0, 2)

        .map(v => v.en);

      const options = shuffle([item.en, ...distractors]);

      return { ru: item.ru, correct: item.en, options };

    });

  }


  function renderQuestion() {

    answered = false;

    const q = questions[currentIndex];

    document.getElementById('question-word').textContent = q.ru;

    document.getElementById('current-q').textContent = currentIndex + 1;

    document.getElementById('score').textContent = score;

    document.getElementById('feedback').textContent = '';

    document.getElementById('feedback').className = 'feedback';

    document.getElementById('next-btn').classList.remove('visible');


    const progress = ((currentIndex) / TOTAL_QUESTIONS) * 100;

    document.getElementById('progress').style.width = progress + '%';


    const optionsEl = document.getElementById('options');

    optionsEl.innerHTML = '';

    q.options.forEach(opt => {

      const btn = document.createElement('button');

      btn.className = 'option-btn';

      btn.textContent = opt;

      btn.onclick = () => selectOption(btn, opt, q.correct);

      optionsEl.appendChild(btn);

    });

  }


  function selectOption(btn, selected, correct) {

    if (answered) return;

    answered = true;


    const allBtns = document.querySelectorAll('.option-btn');

    allBtns.forEach(b => {

      b.disabled = true;

      if (b.textContent === correct) b.classList.add('correct');

    });


    const feedback = document.getElementById('feedback');

    if (selected === correct) {

      score++;

      document.getElementById('score').textContent = score;

      feedback.textContent = '✅ Отлично! Правильно!';

      feedback.className = 'feedback show-correct';

    } else {

      btn.classList.add('wrong');

      feedback.textContent = `❌ Правильный ответ: ${correct}`;

      feedback.className = 'feedback show-wrong';

    }


    document.getElementById('next-btn').classList.add('visible');

  }


  function nextQuestion() {

    currentIndex++;

    if (currentIndex >= TOTAL_QUESTIONS) {

      showResults();

    } else {

      renderQuestion();

    }

  }


  function showResults() {

    document.getElementById('game-screen').classList.add('hidden');

    document.getElementById('result-screen').classList.remove('hidden');

    document.getElementById('result-score').textContent = `${score} / ${TOTAL_QUESTIONS}`;


    const percent = (score / TOTAL_QUESTIONS) * 100;

    let emoji, message;

    if (percent === 100) { emoji = '🏆'; message = 'Идеально! Ты настоящий мастер лексики!'; }

    else if (percent >= 80) { emoji = '🎉'; message = 'Превосходно! Отличный результат!'; }

    else if (percent >= 60) { emoji = '👍'; message = 'Хороший результат! Продолжай в том же духе!'; }

    else if (percent >= 40) { emoji = '💪'; message = 'Неплохо, но есть куда расти!'; }

    else { emoji = '📚'; message = 'Стоит повторить слова. Ты справишься!'; }


    document.getElementById('result-emoji').textContent = emoji;

    document.getElementById('result-message').textContent = message;

  }


  function restartQuiz() {

    currentIndex = 0;

    score = 0;

    questions = generateQuestions();

    document.getElementById('total-q').textContent = TOTAL_QUESTIONS;

    document.getElementById('result-screen').classList.add('hidden');

    document.getElementById('game-screen').classList.remove('hidden');

    renderQuestion();

  }


  document.getElementById('next-btn').onclick = nextQuestion;


  // Запуск

  questions = generateQuestions();

  document.getElementById('total-q').textContent = TOTAL_QUESTIONS;

  renderQuestion();

</script>
